// ============= Config =============
const GAS_URL_KEY = 'gas_proxy_url';
function getGasUrl() { return localStorage.getItem(GAS_URL_KEY) || ''; }
function setGasUrl(u) { localStorage.setItem(GAS_URL_KEY, u); }

// ============= State =============
let currentDate = new Date();
currentDate.setHours(0, 0, 0, 0);
let eventsCache = [];

// ============= Utils =============
const pad = n => (n < 10 ? '0' + n : '' + n);
const fmtDate = d => `${d.getFullYear()}-${pad(d.getMonth()+1)}-${pad(d.getDate())}`;
const weekdayJP = d => '日月火水木金土'[d.getDay()];
function formatTime(iso) {
  if (!iso) return '';
  const d = new Date(iso);
  if (isNaN(d.getTime())) return '';
  return pad(d.getHours()) + ':' + pad(d.getMinutes());
}
function normalizeChars(s) {
  if (!s) return '';
  s = s.replace(/[０-９]/g, c => String.fromCharCode(c.charCodeAt(0) - 0xFEE0));
  s = s.replace(/[：・]/g, ':');
  s = s.replace(/　/g, ' ');
  return s;
}
function localISO(date, hhmm) {
  const [h, m] = hhmm.split(':').map(Number);
  return `${fmtDate(date)}T${pad(h)}:${pad(m)}:00`;
}
function addDays(dateKey, n) {
  const d = parseDateKey(dateKey);
  d.setDate(d.getDate() + n);
  return fmtDate(d);
}
function parseDateKey(key) {
  const [y, m, d] = key.split('-').map(Number);
  return new Date(y, m - 1, d);
}
function showToast(msg) {
  const t = document.getElementById('toast');
  t.textContent = msg;
  t.classList.add('show');
  setTimeout(() => t.classList.remove('show'), 1800);
}

// ============= GAS API =============
async function gasCall(action, payload) {
  const url = getGasUrl();
  if (!url) throw new Error('GAS URLが未設定です');
  const body = JSON.stringify(Object.assign({ action }, payload || {}));
  // text/plain にしてプリフライト回避
  const res = await fetch(url, {
    method: 'POST',
    headers: { 'Content-Type': 'text/plain;charset=utf-8' },
    body
  });
  if (!res.ok) throw new Error('HTTP ' + res.status);
  const data = await res.json();
  if (data && data.error) throw new Error(data.error);
  return data;
}

// ============= Event helpers =============
function isAllDay(ev) {
  if (ev.allDay === true) return true;
  return !!(ev.start && ev.start.date && !ev.start.dateTime);
}
function getEventDateKey(ev) {
  if (ev.start && ev.start.date) return String(ev.start.date).split('T')[0];
  if (ev.start && ev.start.dateTime) return String(ev.start.dateTime).split('T')[0];
  return '';
}
function getStartISO(ev) { return ev.start && ev.start.dateTime ? ev.start.dateTime : null; }
function getEndISO(ev) { return ev.end && ev.end.dateTime ? ev.end.dateTime : null; }
function isDone(ev) { return (ev.summary || '').startsWith('✔️'); }
function plainTitle(ev) {
  const s = ev.summary || '';
  return s.startsWith('✔️') ? s.replace(/^✔️\s*/, '') : s;
}

function sortEvents(events) {
  return events.slice().sort((a, b) => {
    const aDone = isDone(a) ? 1 : 0;
    const bDone = isDone(b) ? 1 : 0;
    if (aDone !== bDone) return aDone - bDone;
    const aAll = isAllDay(a) ? 1 : 0;
    const bAll = isAllDay(b) ? 1 : 0;
    if (aAll !== bAll) return aAll - bAll;
    if (!aAll) {
      const aS = getStartISO(a) || '';
      const bS = getStartISO(b) || '';
      return aS.localeCompare(bS);
    }
    return (a.summary || '').localeCompare(b.summary || '');
  });
}

function getProgressLog(eventId) {
  return localStorage.getItem('progress_' + eventId) || '';
}
function appendProgressLog(eventId, text) {
  const cur = getProgressLog(eventId);
  const now = new Date();
  const stamp = pad(now.getHours()) + ':' + pad(now.getMinutes());
  const line = '[' + stamp + '] ' + text;
  const next = cur ? cur + '\n' + line : line;
  localStorage.setItem('progress_' + eventId, next);
  return next;
}

// ============= Heuristic time parser =============
function parseScheduleText(text) {
  // 入力テキストから予定候補を抽出（正規表現ベース）
  const norm = normalizeChars(text);
  const lines = norm.split(/\r?\n/).map(s => s.trim()).filter(Boolean);
  const results = [];
  for (const line of lines) {
    if (line.length > 200) continue;
    const item = parseLine(line);
    if (item) results.push(item);
    if (results.length >= 15) break;
  }
  return results;
}

function parseLine(line) {
  // パターン1: HH:MM〜HH:MM または HH時(MM分)?〜HH時(MM分)?
  const rangeRe = /(\d{1,2})\s*[:時]\s*(\d{1,2})?\s*分?\s*[〜~\-ー－–—]\s*(\d{1,2})\s*[:時]\s*(\d{1,2})?\s*分?/;
  // パターン2: HH時半
  const halfRe = /(\d{1,2})\s*時\s*半/;
  // パターン3: 単一 HH:MM / HH時(MM分)?
  const singleRe = /(\d{1,2})\s*[:時]\s*(\d{1,2})?\s*分?/;

  let title = line;
  let start = null, end = null, allDay = false;

  let m = line.match(rangeRe);
  if (m) {
    const sh = +m[1], sm = +(m[2] || 0);
    const eh = +m[3], em = +(m[4] || 0);
    if (validHM(sh, sm) && validHM(eh, em)) {
      start = pad(sh) + ':' + pad(sm);
      end = pad(eh) + ':' + pad(em);
      title = line.replace(m[0], '').trim();
    }
  } else {
    m = line.match(halfRe);
    if (m) {
      const h = +m[1];
      if (validHM(h, 30)) {
        start = pad(h) + ':30';
        end = pad((h + 1) % 24) + ':30';
        title = line.replace(m[0], '').trim();
      }
    } else {
      m = line.match(singleRe);
      if (m) {
        const h = +m[1], mm = +(m[2] || 0);
        if (validHM(h, mm)) {
          start = pad(h) + ':' + pad(mm);
          end = pad((h + 1) % 24) + ':' + pad(mm);
          title = line.replace(m[0], '').trim();
        }
      }
    }
  }

  // 記号類を除去してタイトル整形
  title = title.replace(/^[\-・･\*・ー*]\s*/, '').replace(/^[：:]\s*/, '').trim();
  if (!title) return null;
  if (!start) { allDay = true; start = null; end = null; }
  return { title, allDay, start, end, memo: '' };
}

function validHM(h, m) {
  return Number.isInteger(h) && Number.isInteger(m) && h >= 0 && h < 24 && m >= 0 && m < 60;
}

async function fetchEvents(date) {
  const data = await gasCall('list', { date: fmtDate(date) });
  return data.events || [];
}

function renderEvents() {

  if (events.length === 0) {
    container.innerHTML = '<div class="status">予定はありません</div>';
    return;
  }
  container.innerHTML = '';
  for (const ev of events) container.appendChild(renderCard(ev));
}

function renderCard(ev) {

  const memoLabel = document.createElement('div');

// ============= Mutations =============
async function refresh() {
  const container = document.getElementById('cards');
  if (!getGasUrl()) {
    container.innerHTML =
      '<div class="error-box">' +
      'GAS ウェブアプリ URL が未設定です。<br>右上の ⚙ から設定してください。' +
      '<br><button id="openSettingsBtn">設定を開く</button>' +
      '</div>';
    document.getElementById('openSettingsBtn').addEventListener('click', () => openSheet('settings'));
    return;
  }
  container.innerHTML = '<div class="status">読み込み中…</div>';
  try {
    eventsCache = await fetchEvents(currentDate);
    renderEvents();
  } catch (e) {
    console.error(e);
    container.innerHTML = '<div class="error-box">読み込み失敗：' + (e.message || e) + '</div>';
  }
}

async function saveCardEvent(card, ev, btn) {
  const eventId = ev.id;
  const titleInput = card.querySelector('[data-field="title"]');
  const allDayCb = card.querySelector('[data-field="allDay"]');
  const startInput = card.querySelector('[data-field="start"]');
  const endInput = card.querySelector('[data-field="end"]');
  const memo = card.querySelector('[data-field="description"]');

  const newTitle = (isDone(ev) ? '✔️ ' : '') + titleInput.value.trim();
  const wasAllDay = isAllDay(ev);
  const nowAllDay = allDayCb.checked;
  const dateKey = (card.dataset.date || fmtDate(currentDate)).split('T')[0];
  const dateObj = parseDateKey(dateKey);

  btn.disabled = true;
  const orig = btn.textContent;
  btn.textContent = '保存中…';

  try {
    if (wasAllDay !== nowAllDay) {
      // 終日 ⇄ 時間指定: 削除→再作成
      let created;
      if (nowAllDay) {
        created = await gasCall('create', {
          summary: newTitle,
          startTime: dateKey + 'T00:00:00',
          endTime: addDays(dateKey, 1) + 'T00:00:00',
          allDay: true,
          description: memo.value
        });
      } else {
        created = await gasCall('create', {
          summary: newTitle,
          startTime: localISO(dateObj, startInput.value),
          endTime: localISO(dateObj, endInput.value),
          description: memo.value
        });
      }
      if (!created || !created.id) throw new Error('作成失敗');
      await gasCall('delete', { eventId });
    } else if (nowAllDay) {
      await gasCall('update', {
        eventId,
        summary: newTitle,
        description: memo.value
      });
    } else {
      await gasCall('update', {
        eventId,
        summary: newTitle,
        description: memo.value,
        startTime: localISO(dateObj, startInput.value),
        endTime: localISO(dateObj, endInput.value)
      });
    }
    showToast('保存しました');
    await refresh();
  } catch (e) {
    console.error(e);
    showToast('保存失敗: ' + (e.message || e));
  } finally {
    btn.disabled = false;
    btn.textContent = orig;
  }
}

async function toggleDone(ev, btn) {
  const newDone = !isDone(ev);
  const newTitle = newDone ? '✔️ ' + plainTitle(ev) : plainTitle(ev);
  btn.disabled = true;
  try {
    await gasCall('update', { eventId: ev.id, summary: newTitle });
    showToast(newDone ? '完了にしました' : '未完了に戻しました');
    await refresh();
  } catch (e) {
    showToast('更新失敗');
  } finally {
    btn.disabled = false;
  }
}

async function deleteCardEvent(ev) {
  if (!confirm(`「${plainTitle(ev)}」を削除しますか？`)) return;
  try {
    await gasCall('delete', { eventId: ev.id });
    showToast('削除しました');
    await refresh();
  } catch (e) {
    showToast('削除失敗');
  }
}

function syncTabs() {
  const today = new Date(); today.setHours(0, 0, 0, 0);
  const cur = new Date(currentDate); cur.setHours(0, 0, 0, 0);
  const diff = Math.round((cur.getTime() - today.getTime()) / 86400000);
  document.querySelectorAll('.tab').forEach(t => {
    const off = parseInt(t.dataset.offset, 10);
    t.classList.toggle('active', off === diff);
  });
}

// Swipe-down to close sheets
function attachSwipeClose(sheetId, name) {
  const el = document.getElementById(sheetId);
  let startY = null, trackY = 0;
  el.addEventListener('touchstart', e => {
    if (el.scrollTop > 0) { startY = null; return; }
    startY = e.touches[0].clientY; trackY = 0;
  }, { passive: true });
  el.addEventListener('touchmove', e => {
    if (startY === null) return;
    trackY = e.touches[0].clientY - startY;
    if (trackY > 0) el.style.transform = `translateY(${trackY}px)`;
  }, { passive: true });
  el.addEventListener('touchend', () => {
    if (startY === null) return;
    el.style.transform = '';
    if (trackY > 80) closeSheet(name);
    startY = null;
  });
}
